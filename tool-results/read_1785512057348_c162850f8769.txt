    58→  // Sound feedback helper using Web Audio API
    59→  const playClickSound = () => {
    60→    if (!soundEnabled) return;

    62→      const audioCtx = new (window.AudioContext || (window as any).webkitAudioContext)();
    63→      const osc = audioCtx.createOscillator();
    64→      const gain = audioCtx.createGain();
    65→      osc.type = 'sine';
    66→      osc.frequency.setValueAtTime(580, audioCtx.currentTime);
    67→      osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.08);
    68→      gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    69→      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
    70→      osc.connect(gain);
    71→      gain.connect(audioCtx.destination);
    72→      osc.start();
    73→      osc.stop(audioCtx.currentTime + 0.08);
    74→    } catch (e) {
    75→      // Audio context fallback
    76→    }
    77→  };

    80→

    82→  useEffect(() => {
    83→    const isMobile = window.innerWidth < 768 || /Mobi|Android|iPhone|iPad/i.test(navigator.userAgent);
    84→    if (isMobile) {
    85→      return;
    86→    }
    87→
    88→    const lenis = new Lenis({
    89→      duration: 1.1,
    90→      easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),

    92→      gestureOrientation: 'vertical',
    93→      smoothWheel: true,
    94→    });

    96→
    97→    lenis.on('scroll', ScrollTrigger.update);
    98→

   100→      lenis.raf(time * 1000);
   101→    };
   102→    gsap.ticker.add(tickHandler);
   103→    gsap.ticker.lagSmoothing(0);
   104→
   105→    return () => {
   106→      lenis.off('scroll', ScrollTrigger.update);
   107→      gsap.ticker.remove(tickHandler);
   108→      lenis.destroy();
   109→    };
   110→  }, []);
   111→
   112→  // Lock document body scroll and stop Lenis scroll when any modal or drawer is active

   114→    Boolean(selectedProductDetail) ||

   127→      document.body.style.overflow = '';
   128→      lenisInstance?.start();
   129→    }
   130→    return () => {
   131→      document.body.style.overflow = '';
   132→      lenisInstance?.start();
   133→    };
   134→  }, [isAnyModalOpen, lenisInstance]);

   141→        if (data.products && data.products.length > 0) {
   142→          setProducts(data.products);
   143→        }
   144→      })
   145→      .catch((err) => console.warn('Using local catalog fallback:', err));
   146→  }, []);
   147→
   148→  // Catalog section entrance animation
   149→  useEffect(() => {
   150→    if (catalogRef.current) {
   151→      gsap.fromTo(
   152→        catalogRef.current,
   153→        { opacity: 0, y: 30 },
   154→        {
   155→          opacity: 1,
   156→          y: 0,
   157→          duration: 0.8,
   158→          ease: 'power3.out',
   159→          scrollTrigger: {
   160→            trigger: catalogRef.current,
   161→            start: 'top 85%',
   162→          },
   163→        }
   164→      );
   165→    }
   166→  }, []);
   167→
   168→  // Cart operations
   169→  const handleAddToCart = (product: RiceProduct, weightKg: number) => {
   170→    playClickSound();
   171→    setCartItems((prev) => {
   172→      const existingIndex = prev.findIndex(
   173→        (item) => item.productId === product.id && item.selectedWeightKg === weightKg
   174→      );
   175→
   176→      const basePrice = product.discountedPricePerKg || product.pricePerKg;
   177→      let bulkMultiplier = 1;
   178→      if (weightKg === 10) bulkMultiplier = 0.95;
   179→      if (weightKg === 25) bulkMultiplier = 0.90;
   180→
   181→      const unitPricePerKg = Math.round(basePrice * bulkMultiplier);
   182→
   183→      if (existingIndex > -1) {
   184→        const updated = [...prev];
   185→        const item = updated[existingIndex];
   186→        const newQty = item.quantity + 1;
   187→        updated[existingIndex] = {
   188→          ...item,
   189→          quantity: newQty,
   190→          totalPrice: unitPricePerKg * weightKg * newQty,
   191→        };
   192→        return updated;
   193→      } else {
   194→        return [
   195→          ...prev,
   196→          {
   197→            productId: product.id,
   198→            product,
   199→            selectedWeightKg: weightKg,
   200→            quantity: 1,
   201→            unitPricePerKg,
   202→            totalPrice: unitPricePerKg * weightKg,
   203→          },
   204→        ];
   205→      }
   206→    });
   207→  };
   208→
   209→  const handleUpdateCartQuantity = (productId: string, weightKg: number, newQty: number) => {
   210→    playClickSound();
   211→    if (newQty <= 0) {
   212→      handleRemoveCartItem(productId, weightKg);
   213→      return;
   214→    }
   215→
   216→    setCartItems((prev) =>
   217→      prev.map((item) => {
   218→        if (item.productId === productId && item.selectedWeightKg === weightKg) {
   219→          return {
   220→            ...item,
   221→            quantity: newQty,
   222→            totalPrice: item.unitPricePerKg * weightKg * newQty,
   223→          };
   224→        }
   225→        return item;
   226→      })
   227→    );
   228→  };
   229→
   230→  const handleRemoveCartItem = (productId: string, weightKg: number) => {
   231→    playClickSound();
   232→    setCartItems((prev) =>
   233→      prev.filter((item) => !(item.productId === productId && item.selectedWeightKg === weightKg))
   234→    );
   235→  };

   239→  };

   244→    if (activeCategory !== 'all' && p.category !== activeCategory) {
   245→      return false;
   246→    }
   247→
   248→    // GI Index check
   249→    if (giFilter === 'low' && !p.giIndex.includes('Low')) {
   250→      return false;
   251→    }
   252→

   254→    if (searchQuery.trim()) {
   255→      const q = searchQuery.toLowerCase();
   256→      const matchName = p.name.toLowerCase().includes(q);
   257→      const matchTag = p.tagline.toLowerCase().includes(q);
   258→      const matchOrigin = p.originRegion.toLowerCase().includes(q);
   259→      const matchDish = p.bestFor.some((d) => d.toLowerCase().includes(q));
   260→
   261→      if (!matchName && !matchTag && !matchOrigin && !matchDish) {
   262→        return false;
   263→      }
   264→    }
   265→
   266→    return true;
   267→  });

   272→    const priceB = b.discountedPricePerKg || b.pricePerKg;
   273→
   274→    if (sortBy === 'price-low') return priceA - priceB;
   275→    if (sortBy === 'price-high') return priceB - priceA;
   276→    if (sortBy === 'rating') return b.rating - a.rating;
   277→    return 0; // featured default
   278→  });

   285→      start: 'top 90%',
   286→      onEnter: () => {
   287→        gsap.fromTo(
   288→          '.catalog-filter-bar',
   289→          { opacity: 0, y: 30, scale: 0.98 },
   290→          { opacity: 1, y: 0, scale: 1, duration: 0.85, ease: 'power4.out', overwrite: 'auto' }
   291→        );
   292→      },
   293→      once: true,
   294→    });
   295→
   296→    // 2. Category Strip Reveal (if present)
   297→    const categoryTrigger = ScrollTrigger.create({
   298→      trigger: '.catalog-category-strip',
   299→      start: 'top 92%',
   300→      onEnter: () => {
   301→        gsap.fromTo(
   302→          '.catalog-category-strip',
   303→          { opacity: 0, y: 20 },
   304→          { opacity: 1, y: 0, duration: 0.75, ease: 'power4.out', overwrite: 'auto' }
   305→        );
   306→      },
   307→      once: true,
   308→    });
   309→
   310→    // 3. Grid Cards Entrance Animation
   311→    let batch: any[] = [];
   312→    if (!isLoading && gridRef.current) {
   313→      const cards = Array.from(gridRef.current.children) as HTMLElement[];
   314→      if (cards.length > 0) {
   315→        gsap.killTweensOf(cards);
   316→        gsap.set(cards, { opacity: 0, y: 40, scale: 0.96 });
   317→
   318→        batch = ScrollTrigger.batch(cards, {
   319→          start: 'top 95%',
   320→          onEnter: (batchElements) => {
   321→            gsap.to(batchElements, {
   322→              opacity: 1,
   323→              y: 0,
   324→              scale: 1,
   325→              duration: 0.85,
   326→              ease: 'power4.out',
   327→              stagger: 0.06,
   328→              overwrite: 'auto',
   329→            });
   330→          },
   331→          once: true,
   332→        });
   333→      }
   334→    }
   335→
   336→    return () => {
   337→      filterTrigger.kill();
   338→      categoryTrigger.kill();
   339→      batch.forEach((st) => st.kill());
   340→    };
   341→  }, [sortedProducts, isLoading]);
   342→

   392→            playClickSound();
   393→            setComparisonModalOpen(true);
   394→          }}
   395→        />
   396→
   397→        {/* Product Catalog Section */}

   401→            {[
   402→              { id: 'all', label: 'All Varieties', icon: Sprout },
   403→              { id: 'daily', label: 'Daily Staples', icon: Layers },
   404→              { id: 'heritage', label: 'Heritage & Unpolished', icon: Sprout },
   405→              { id: 'aromatic', label: 'Aromatic & Basmati', icon: Sparkles },
   406→              { id: 'superfood', label: 'Diabetic & Low GI', icon: HeartHandshake },
   407→              { id: 'combos', label: 'Value Bundles', icon: Package },
   408→            ].map((cat) => {
   409→              const Icon = cat.icon;
   410→              const isSelected = activeCategory === cat.id;
   411→              return (
   412→                <button

   419→                    isSelected

   476→                  giFilter === 'low'
   477→                    ? 'bg-[#1F431E] text-white border-[#1F431E] shadow-xs'
   478→                    : 'bg-[#1F431E]/5 text-[#1F431E] border-[#1F431E]/20 hover:bg-[#1F431E]/10'

   481→                <Filter className="w-3.5 h-3.5" />

   496→                >
   497→                  <option value="featured">Featured / Best Seller</option>
   498→                  <option value="price-low">Price: Low to High</option>
   499→                  <option value="price-high">Price: High to Low</option>
   500→                  <option value="rating">Customer Rating</option>
   501→                </select>

   503→            </div>
   504→          </div>
   505→
   506→          {/* Product Grid */}

   512→            <div className="text-center py-16 bg-white rounded-2xl border border-stone-200 p-8 space-y-4">
   513→              <div className="w-16 h-16 rounded-full bg-[#2D5A27]/10 flex items-center justify-center mx-auto text-[#2D5A27]">
   514→                <Sprout className="w-8 h-8" />
   515→              </div>
   516→              <h3 className="text-lg font-serif text-stone-800">No matching rice grains found</h3>
   517→              <p className="text-xs text-stone-500 max-w-sm mx-auto">
   518→                Try clearing your search query or reset filters to view all available organic grains.
   519→              </p>
   520→              <button
   521→                onClick={() => {
   522→                  playClickSound();
   523→                  setSearchQuery('');
   524→                  setActiveCategory('all');
   525→                  setGiFilter('all');
   526→                }}
   527→                className="px-4 py-2 bg-[#2D5A27] text-white rounded-xl text-xs font-bold hover:bg-[#23471e]"
   528→              >
   529→                Reset Catalog Filters

   541→                    setSelectedProductDetail(p);

   545→            </div>
   546→          )}
   547→        </section>
   548→
   549→        {/* Grain Wisdom & Organic Heritage Hub */}
   550→        <GrainWisdomHub />
   551→      </main>
   552→
   553→      {/* Footer */}
   554→      <Footer
   555→        onOpenSheetsModal={() => {
   556→          playClickSound();
