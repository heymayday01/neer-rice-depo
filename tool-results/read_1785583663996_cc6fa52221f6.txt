   219→function lookupPincode(pin: string): { city: string; state: string } | null {
   220→  if (pin.length < 2) return null;
   221→  return PINCODE_MAP[pin.slice(0, 2)] || null;
   222→}
   223→
   224→function detectCardType(number: string): string | null {
   225→  const n = number.replace(/\s/g, "");
   226→  if (!n) return null;
   227→  if (/^4/.test(n)) return "VISA";
   228→  if (/^(5[1-5]|2[2-7])/.test(n)) return "Mastercard";
   229→  if (/^3[47]/.test(n)) return "Amex";
   230→  if (/^(60|65|81|82)/.test(n)) return "RuPay";
   231→  return null;
   232→}

   234→function addDays(date: Date, days: number): Date {
   235→  const d = new Date(date);
   236→  d.setDate(d.getDate() + days);
   237→  return d;

   241→  const opts: Intl.DateTimeFormatOptions = { month: "short", day: "numeric" };
